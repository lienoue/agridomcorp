    from django.contrib.auth import get_user_model from datetime import date User = get_user_model() today = date.today() for u in User.objects.filter(birth_date__isnull=False, mode_junior=False): age = today.year - u.birth_date.year - ( (today.month, today.day) < (u.birth_date.month, u.birth_date.day) ) if age < 18: User.objects.filter(pk=u.pk).update(mode_junior=True) print(f"✅ {u.username} ({age} ans)")
